t = int(input())
for _ in range(t):
    n = int(input())
    string = input()
    x = 0
    y = 0
    for i in string:
        if i == "U":
            x += 1
        elif i == "D":
            x -= 1
        elif i == "L":
            y -= 1
        elif i == "R":
            y += 1
        if x == 1 and y == 1:
            print("YES")
            break
    else:
        print("NO")
