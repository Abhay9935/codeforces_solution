t = int(input())
for j in range(t):
    n = int(input())
    s = input()
    x = 0
    y = 0
    found = False
    for i in s:
        if i == 'L':
            x -= 1
        elif i == 'R':
            x += 1
        elif i == 'U':
            y += 1
        elif i == 'D':
            y -= 1
        if x == 1 and y == 1:
            found = True
            break
    if found == True:
        print("YES")
    else:
        print("NO")
