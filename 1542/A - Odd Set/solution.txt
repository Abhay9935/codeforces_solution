t = int(input())
for i in range(t):
    n = int(input())
    numbers = list(map(int, input().split()))
    even_count = 0
    odd_count = 0
    for i in range(2*n):
        if numbers[i] % 2 == 0:
            even_count += 1
        else:
            odd_count += 1
    if even_count == odd_count:
        print("YES")
    else:
        print("NO")
