t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    a.sort()
    left = []
    middle = []
    right = []
    i = 0
    while i < n:
        j = i
        while j < n and a[j] == a[i]:
            j += 1
        count = j - i
        left.append(a[i])
        if count >= 2:
            right.append(a[i])
        for _ in range(count - 2):
            middle.append(a[i])
        i = j
    right.reverse()
    k = len(left) + len(right)
    print(k)
    total = left + middle + right
    print(*total)
