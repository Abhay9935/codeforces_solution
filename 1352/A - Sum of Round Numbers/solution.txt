t = int(input())
for j in range(t):
    number = int(input())
    count = 0
    i = 0
    ans = []
    while number > 0:
        digit = number % 10
        number = number // 10
        if digit == 0:
            count += 0
        else:
            count += 1
            ans.append(digit * 10**i)
        i += 1
 
    print(count)
    print(*ans)
