t = int(input())
for i in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    count = 0
    negative_count = 0
    for i in range(n):
        if a[i] < 0:
            negative_count += 1
 
    if negative_count % 2 == 0:
        for i in range(n):
            if a[i] == 0:
                count += 1
    else:
        for i in range(1, 3):
            for i in range(n):
                if a[i] == 0:
                    a[i] += 1
                    count += 1
            else:
                a[i] > 0
                count += 1
    print(count)
